cmake_minimum_required(VERSION 3.20)
project(MMSCurveRepair VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build this plugin with MSVC for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /EHsc /fp:precise)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
set(MMS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(repair_core STATIC repair_core.cpp "${MMS_ROOT}/src/animation/animation_curve.cpp")
target_include_directories(repair_core PUBLIC . "${MMS_ROOT}/include")
add_library(curve_repair SHARED plugin.cpp)
set_target_properties(curve_repair PROPERTIES OUTPUT_NAME MMSCurveRepair)
target_link_libraries(curve_repair PRIVATE repair_core comctl32 user32 gdi32)
add_executable(curve_repair_launcher WIN32 launcher.cpp)
set_target_properties(curve_repair_launcher PROPERTIES OUTPUT_NAME MMSCurveRepairLauncher)
target_link_libraries(curve_repair_launcher PRIVATE comdlg32 comctl32 user32 gdi32 psapi)
add_executable(repair_tests tests.cpp)
target_link_libraries(repair_tests PRIVATE repair_core)
enable_testing()
add_test(NAME repair_core COMMAND repair_tests)
