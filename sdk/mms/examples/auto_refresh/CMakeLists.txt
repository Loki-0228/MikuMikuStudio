cmake_minimum_required(VERSION 3.21)
project(mms_auto_refresh LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MMS ABI 1 requires Windows x64")
endif()
add_library(auto_refresh SHARED plugin.cpp plugin.def)
target_include_directories(auto_refresh PRIVATE ../../include)
target_compile_features(auto_refresh PRIVATE cxx_std_17)
target_compile_definitions(auto_refresh PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601)
if(MSVC)
    target_compile_options(auto_refresh PRIVATE /utf-8)
    target_link_options(auto_refresh PRIVATE /SUBSYSTEM:WINDOWS,6.01)
    set_target_properties(auto_refresh PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_custom_command(TARGET auto_refresh POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/plugin.json" "$<TARGET_FILE_DIR:auto_refresh>/plugin.json")
