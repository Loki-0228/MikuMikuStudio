cmake_minimum_required(VERSION 3.21)
project(mms_hello_menu LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MMS ABI 1 requires Windows x64")
endif()
add_library(hello_menu SHARED plugin.cpp plugin.def)
target_include_directories(hello_menu PRIVATE ../../include)
target_compile_features(hello_menu PRIVATE cxx_std_17)
target_compile_definitions(hello_menu PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601)
if(MSVC)
    target_compile_options(hello_menu PRIVATE /utf-8)
    target_link_options(hello_menu PRIVATE /SUBSYSTEM:WINDOWS,6.01)
    set_target_properties(hello_menu PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_custom_command(TARGET hello_menu POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/plugin.json" "$<TARGET_FILE_DIR:hello_menu>/plugin.json")
